# One-shot trial boot helper for the A733 OTA workflow.
#
# The parent boot script sources this only after it has prepared bootargs.
# A pending trial is allowed exactly once. The state lives in raw MMC sector
# 65520 (0xfff0), which is inside the verified unused gap before mmcblk0p1.
# U-Boot changes pending to attempted before loading the kernel, without ever
# modifying the journaled ext4 root filesystem.

if mmc dev 0; then
	if mmc read ${load_addr} 0xfff0 1; then
		if itest.b *${load_addr} == 0xa7; then
			setexpr ota_magic_addr ${load_addr} + 1
			if itest.b *${ota_magic_addr} == 0x33; then
				setexpr ota_state_addr ${load_addr} + 2
				if itest.b *${ota_state_addr} == 0x02; then
					echo "A733 OTA: prior trial did not commit; loading rollback boot set"
					if load ${devtype} ${devnum} ${kernel_addr_r} ${prefix}a733-ota/rollback/Image; then
						if load ${devtype} ${devnum} ${fdt_addr_r} ${prefix}a733-ota/rollback/sun60i-a733-orangepi-4-pro.dtb; then
							setenv bootargs "${bootargs} a733.ota_rollback=1"
							fdt addr ${fdt_addr_r}
							fdt resize 65536
							booti ${kernel_addr_r} - ${fdt_addr_r}
						fi
					fi
					echo "A733 OTA: rollback files unavailable; continuing normal boot"
				fi
				if itest.b *${ota_state_addr} == 0x01; then
					echo "A733 OTA: marking and booting trial set"
					mw.b ${load_addr} 0 0x200
					mw.b ${load_addr} 0xa7 1
					setexpr ota_magic_addr ${load_addr} + 1
					mw.b ${ota_magic_addr} 0x33 1
					setexpr ota_state_addr ${load_addr} + 2
					mw.b ${ota_state_addr} 0x02 1
					setexpr ota_check_addr ${load_addr} + 3
					mw.b ${ota_check_addr} 0x5a 1
					if mmc write ${load_addr} 0xfff0 1; then
						if load ${devtype} ${devnum} ${kernel_addr_r} ${prefix}a733-ota/trial/Image; then
							if load ${devtype} ${devnum} ${fdt_addr_r} ${prefix}a733-ota/trial/sun60i-a733-orangepi-4-pro.dtb; then
								setenv bootargs "${bootargs} a733.ota_trial=1"
								fdt addr ${fdt_addr_r}
								fdt resize 65536
								booti ${kernel_addr_r} - ${fdt_addr_r}
							fi
						fi
						echo "A733 OTA: trial files unavailable; continuing normal boot"
					else
						echo "A733 OTA: cannot persist raw trial guard; continuing normal boot"
					fi
				fi
			fi
		fi
	fi
fi
